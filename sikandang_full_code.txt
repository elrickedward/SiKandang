            'nama_aturan' => 'required|string|max:255',

            'operator' => 'required|string|max:20',
            'nilai_ambang' => 'nullable|numeric',
            'nilai_ambang_atas' => 'nullable|numeric',
            'kata_kunci_kondisi' => 'nullable|string|max:255',
            'saran_tindakan' => 'required|string',
            'prioritas' => 'required|in:Segera,Waspada,Aman',
        ]);

        AturanRekomendasi::create(array_merge(
            $request->all(),
            ['user_id' => Auth::id(), 'aktif' => true]
        ));

        return back()->with('success', 'Aturan rekomendasi baru berhasil ditambahkan.');
    }

    public function update(Request $request, AturanRekomendasi $aturan)
    {
        $request->validate([
            'nama_aturan' => 'required|string|max:255',
            'nilai_ambang' => 'nullable|numeric',
            'nilai_ambang_atas' => 'nullable|numeric',
            'kata_kunci_kondisi' => 'nullable|string|max:255',
            'saran_tindakan' => 'required|string',
            'prioritas' => 'required|in:Segera,Waspada,Aman',
        ]);

        $aturan->update([
            'nama_aturan' => $request->nama_aturan,
            'nilai_ambang' => $request->nilai_ambang,
            'nilai_ambang_atas' => $request->nilai_ambang_atas,

            'saran_tindakan' => $request->saran_tindakan,
            'prioritas' => $request->prioritas,
            'aktif' => $request->has('aktif'),
        ]);

        return back()->with('success', "Aturan {$aturan->kode_aturan} berhasil diperbarui.");
    }

<?php

namespace App\Http\Controllers;

use App\Models\DetailPesanan;
use App\Models\Pembeli;
use App\Models\Pesanan;
use App\Models\Produk;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class BerandaController extends Controller
{
    public function index()
    {
        $produks = Produk::orderBy('nama_produk')->get();
        return view('beranda', compact('produks'));
    }

    public function pesan(Request $request)
    {
        $request->validate([
            'nama' => 'required|string|max:255',
            'no_wa' => 'required|string|max:25',
            'alamat' => 'required|string',
            'metode_pembayaran' => 'required|string|max:100',
            'items' => 'required|array',
        ]);

        // Filter produk dengan jumlah > 0
        $itemDipilih = collect($request->items)->filter(fn ($qty) => (int) $qty > 0);

        if ($itemDipilih->isEmpty()) {
            return back()->with('error', 'Silakan pilih minimal 1 produk dengan jumlah lebih dari 0.');
        }

        try {
            $pesanan = DB::transaction(function () use ($request, $itemDipilih) {
                $totalHarga = 0;
                $rincian = [];

                foreach ($itemDipilih as $produkId => $qty) {
                    $qty = (int) $qty;
                    $produk = Produk::lockForUpdate()->findOrFail($produkId);

                    // Validasi sisi server: tolak jika jumlah melebihi stok
                    if ($qty > $produk->stok) {
                        throw new \Exception("Jumlah pesanan {$produk->nama_produk} ({$qty}) melebihi stok tersedia ({$produk->stok}).");
                    }

                    $subtotal = $produk->harga_satuan * $qty;
                    $totalHarga += $subtotal;

                    $rincian[] = [
                        'produk' => $produk,
                        'jumlah' => $qty,
                        'harga_satuan' => $produk->harga_satuan,
                        'subtotal' => $subtotal,
                    ];
                }

                $pembeli = Pembeli::create([
                    'nama' => $request->nama,
                    'no_wa' => $request->no_wa,
                    'alamat' => $request->alamat,
                ]);

                $pesanan = Pesanan::create([
                    'pembeli_id' => $pembeli->id,
                    'nomor_pesanan' => 'SK-' . date('Ymd') . '-' . strtoupper(Str::random(4)),
                    'tanggal_pesanan' => now(),
                    'total_harga' => $totalHarga,
                    'metode_pembayaran' => $request->metode_pembayaran,
                    'status' => 'Baru',
                ]);

                foreach ($rincian as $item) {
                    DetailPesanan::create([
                        'pesanan_id' => $pesanan->id,
                        'produk_id' => $item['produk']->id,
                        'jumlah' => $item['jumlah'],
                        'harga_satuan' => $item['harga_satuan'],
                        'subtotal' => $item['subtotal'],
                    ]);

                    // Kurangi stok produk
                    $item['produk']->decrement('stok', $item['jumlah']);
                }

                return $pesanan;
            });

            return redirect()->route('beranda')->with('pesanan_sukses', $pesanan->load('pembeli', 'detailPesanan.produk'));
        } catch (\Exception $e) {
            return back()->withInput()->with('error', $e->getMessage());
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\CatatanHarian;
use App\Models\Kandang;
use App\Models\Pakan;
use App\Models\RiwayatKesehatan;
use App\Services\RekomendasiService;
use Illuminate\Http\Request;

class CatatanHarianController extends Controller
{
    public function index()
    {
        $kandangs = Kandang::all();
        $pakans = Pakan::all();
        $catatans = CatatanHarian::with('kandang', 'pakan')->latest('tanggal')->latest('id')->get();
        $riwayatKesehatan = RiwayatKesehatan::with('kandang')->latest('tanggal')->latest('id')->get();

        return view('pemilik.catatan.index', compact('kandangs', 'pakans', 'catatans', 'riwayatKesehatan'));
    }

    public function store(Request $request, RekomendasiService $dss)
    {
        $request->validate([
            'kandang_id' => 'required|exists:kandang,id',
            'pakan_id' => 'required|exists:pakan,id',
            'tanggal' => 'required|date',
            'jumlah_ayam' => 'required|integer|min:0',
            'kematian' => 'required|integer|min:0',
            'pakan_terpakai' => 'required|numeric|min:0',
            'kondisi' => 'required|string|max:255',
        ]);

        $pakan = Pakan::findOrFail($request->pakan_id);

        // Validasi sisi server: tolak jika pakan terpakai melebihi stok tersedia
        if ($request->pakan_terpakai > $pakan->stok_saat_ini) {
            return back()->withInput()->with('error', "Gagal! Pakan terpakai ({$request->pakan_terpakai} {$pakan->satuan}) melebihi sisa stok ({$pakan->stok_saat_ini} {$pakan->satuan}).");
        }

        CatatanHarian::create($request->only('kandang_id', 'pakan_id', 'tanggal', 'jumlah_ayam', 'kematian', 'pakan_terpakai', 'kondisi'));
        $pakan->decrement('stok_saat_ini', $request->pakan_terpakai);

        // Jalankan evaluasi aturan DSS
        $dss->evaluasiSemua($request->kandang_id);

        return back()->with('success', 'Catatan harian tersimpan, stok pakan dikurangi, dan rekomendasi diperbarui.');
    }

    public function storeKesehatan(Request $request, RekomendasiService $dss)
    {
        $request->validate([
            'kandang_id' => 'required|exists:kandang,id',
            'tanggal' => 'required|date',
            'gejala_penyakit' => 'required|string|max:255',
            'tindakan' => 'required|string',
        ]);

        $riwayat = RiwayatKesehatan::create($request->only('kandang_id', 'tanggal', 'gejala_penyakit', 'tindakan'));
        $dss->evaluasiRiwayatPenyakit($riwayat);

        return back()->with('success', 'Riwayat kesehatan berhasil dicatat.');
    }
}

namespace App\Http\Controllers;

use App\Models\Pesanan;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class PesananController extends Controller
{
    public function index()
    {
        $pesanans = Pesanan::with('pembeli', 'detailPesanan.produk')
            ->latest('tanggal_pesanan')
            ->get();

        return view('pemilik.pesanan.index', compact('pesanans'));
    }

    public function updateStatus(Request $request, Pesanan $pesanan)
    {
        $request->validate([
            'status' => 'required|in:Baru,Diproses,Selesai,Dibatalkan',
        ]);

        // Kembalikan stok produk jika pesanan dibatalkan
        if ($request->status === 'Dibatalkan' && $pesanan->status !== 'Dibatalkan') {
            foreach ($pesanan->detailPesanan as $detail) {
                $detail->produk->increment('stok', $detail->jumlah);
            }
        }

        $pesanan->update([
            'status' => $request->status,
            'user_id' => Auth::id(),
        ]);

        return back()->with('success', "Status pesanan {$pesanan->nomor_pesanan} diubah menjadi {$request->status}.");
    }
}

<?php

namespace App\Http\Controllers;

use App\Http\Requests\ProfileUpdateRequest;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Redirect;
use Illuminate\View\View;

class ProfileController extends Controller
{
    /**
     * Display the user's profile form.
     */
    public function edit(Request $request): View
    {
        return view('profile.edit', [
            'user' => $request->user(),
        ]);
    }

    /**
     * Update the user's profile information.
     */
    public function update(ProfileUpdateRequest $request): RedirectResponse
    {
        $request->user()->fill($request->validated());

        if ($request->user()->isDirty('email')) {
            $request->user()->email_verified_at = null;
        }

        $request->user()->save();

        return Redirect::route('profile.edit')->with('status', 'profile-updated');
    }

    /**
     * Delete the user's account.
     */
    public function destroy(Request $request): RedirectResponse
    {
        $request->validateWithBag('userDeletion', [
            'password' => ['required', 'current_password'],
        ]);

        $user = $request->user();

        Auth::logout();

        $user->delete();

        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return Redirect::to('/');
    }
}

namespace App\Http\Controllers;

use App\Models\Produk;

use App\Models\SimulasiBiaya;
use App\Services\RekomendasiService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class RekomendasiController extends Controller
{
    public function index(RekomendasiService $dss)
    {
        $dss->evaluasiSemua();

        $rekomendasis = Rekomendasi::with('kandang')
            ->where('status', 'aktif')
            ->orderByRaw("FIELD(prioritas, 'Segera', 'Waspada', 'Aman')")
            ->latest('tanggal')
            ->get();

        $produks = Produk::all();
        $simulasiTerbaru = SimulasiBiaya::latest()->take(5)->get();

        return view('pemilik.rekomendasi.index', compact('rekomendasis', 'produks', 'simulasiTerbaru'));
    }

    public function tandaiSelesai(Rekomendasi $rekomendasi)
    {
        $rekomendasi->update(['status' => 'selesai']);
        return back()->with('success', 'Rekomendasi ditandai selesai.');
    }

    public function simulasi(Request $request)
    {
        $request->validate([
            'nama_simulasi' => 'required|string|max:255',
            'rencana_pakan' => 'required|numeric|min:0',
            'harga_pakan' => 'required|numeric|min:0',
            'ayam_dijual' => 'required|integer|min:1',
            'harga_jual' => 'required|numeric|min:0',
        ]);

        $estBiaya = $request->rencana_pakan * $request->harga_pakan;
        $estPendapatan = $request->ayam_dijual * $request->harga_jual;
        $estLaba = $estPendapatan - $estBiaya;

        $hasil = SimulasiBiaya::create([
            'user_id' => Auth::id(),
            'nama_simulasi' => $request->nama_simulasi,
            'tanggal' => now()->toDateString(),
            'rencana_pakan' => $request->rencana_pakan,
            'harga_pakan' => $request->harga_pakan,
            'ayam_dijual' => $request->ayam_dijual,
            'harga_jual' => $request->harga_jual,
            'est_biaya' => $estBiaya,
            'est_pendapatan' => $estPendapatan,
            'est_laba' => $estLaba,
        ]);

        return back()->with('hasil_simulasi', $hasil);
    }
}

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\ValidationException;
use Illuminate\View\View;

class ConfirmablePasswordController extends Controller
{
    /**
     * Show the confirm password view.
     */
    public function show(): View
    {
        return view('auth.confirm-password');
    }

    /**
     * Confirm the user's password.
     */
    public function store(Request $request): RedirectResponse
    {
        if (! Auth::guard('web')->validate([
            'email' => $request->user()->email,
            'password' => $request->password,
        ])) {
            throw ValidationException::withMessages([
                'password' => __('auth.password'),
            ]);
        }

        $request->session()->put('auth.password_confirmed_at', time());

        return redirect()->intended(route('dashboard', absolute: false));
    }
}

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Auth\Events\PasswordReset;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Password;
use Illuminate\Support\Str;
use Illuminate\Validation\Rules;
use Illuminate\Validation\ValidationException;
use Illuminate\View\View;

class NewPasswordController extends Controller
{
    /**
     * Display the password reset view.
     */
    public function create(Request $request): View
    {
        return view('auth.reset-password', ['request' => $request]);
    }

    /**
     * Handle an incoming new password request.
     *
     * @throws ValidationException
     */
    public function store(Request $request): RedirectResponse
    {
        $request->validate([
            'token' => ['required'],
            'email' => ['required', 'email'],
            'password' => ['required', 'confirmed', Rules\Password::defaults()],
        ]);

        // Here we will attempt to reset the user's password. If it is successful we
        // will update the password on an actual user model and persist it to the
        // database. Otherwise we will parse the error and return the response.
        $status = Password::reset(
            $request->only('email', 'password', 'password_confirmation', 'token'),
            function (User $user) use ($request) {
                $user->forceFill([
                    'password' => Hash::make($request->password),
                    'remember_token' => Str::random(60),
                ])->save();

                event(new PasswordReset($user));
            }
        );

        // If the password was successfully reset, we will redirect the user back to
        // the application's home authenticated view. If there is an error we can
        // redirect them back to where they came from with their error message.
        return $status == Password::PASSWORD_RESET
                    ? redirect()->route('login')->with('status', __($status))
                    : back()->withInput($request->only('email'))
                        ->withErrors(['email' => __($status)]);
    }
}

<?php

namespace App\Http\Requests\Auth;

use Illuminate\Auth\Events\Lockout;
use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class LoginRequest extends FormRequest
{
    /**
     * Determine if the user is authorized to make this request.
     */
    public function authorize(): bool
    {
        return true;
    }

    /**
     * Get the validation rules that apply to the request.
     *
     * @return array<string, ValidationRule|array<mixed>|string>
     */
    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ];
    }

    /**
     * Attempt to authenticate the request's credentials.
     *
     * @throws ValidationException
     */
    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        if (! Auth::attempt($this->only('email', 'password'), $this->boolean('remember'))) {
            RateLimiter::hit($this->throttleKey());

            throw ValidationException::withMessages([
                'email' => trans('auth.failed'),
            ]);
        }

        RateLimiter::clear($this->throttleKey());
    }

    /**
     * Ensure the login request is not rate limited.
     *
     * @throws ValidationException
     */
    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        event(new Lockout($this));

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'email' => trans('auth.throttle', [
                'seconds' => $seconds,
                'minutes' => ceil($seconds / 60),
            ]),
        ]);
    }

    /**
     * Get the rate limiting throttle key for the request.
     */
    public function throttleKey(): string
    {
        return Str::transliterate(Str::lower($this->string('email')).'|'.$this->ip());
    }
}

<?php

namespace App\Services;

use App\Models\AturanRekomendasi;
use App\Models\CatatanHarian;
use App\Models\Kandang;
use App\Models\Pakan;
use App\Models\Rekomendasi;
use App\Models\RiwayatKesehatan;
use App\Models\SimulasiBiaya;
use Carbon\Carbon;

class RekomendasiService
{
    /**
     * Evaluasi seluruh aturan pakan dan kesehatan untuk sebuah kandang
     */
    public function evaluasiSemua(?int $kandangId = null): void
    {
        $kandang = $kandangId ? Kandang::find($kandangId) : Kandang::first();
        if (!$kandang) {
            return;
        }

        $this->evaluasiPakan($kandang);
        $this->evaluasiKesehatan($kandang);
    }

    /**
     * Evaluasi Aturan Pakan (R1, R2, R3, dan aturan pakan kustom)
     */
    public function evaluasiPakan(Kandang $kandang): void
    {
        $aturanPakan = AturanRekomendasi::where('kategori', 'pakan')
            ->where('aktif', true)
            ->get();

        foreach (Pakan::all() as $pakan) {
            // Total pakan terpakai 7 hari terakhir
            $tujuhHariLalu = Carbon::today()->subDays(6);
            $total7Hari = CatatanHarian::where('pakan_id', $pakan->id)
                ->where('tanggal', '>=', $tujuhHariLalu)
                ->sum('pakan_terpakai');

            // Rata-rata pakan harian = total 7 hari / 7 (atau fallback ke pemakaian terakhir jika belum 7 hari)
            $rataRataHarian = $total7Hari > 0 ? round($total7Hari / 7, 2) : 0;
            if ($rataRataHarian <= 0) {
                $catatanTerakhir = CatatanHarian::where('pakan_id', $pakan->id)->latest('tanggal')->first();
                $rataRataHarian = $catatanTerakhir ? (float) $catatanTerakhir->pakan_terpakai : 0;
            }

            if ($rataRataHarian <= 0) {
                continue;
            }

            // Sisa hari stok & jumlah pembelian disarankan (target cadangan 10 hari)
            $sisaHariStok = round($pakan->stok_saat_ini / $rataRataHarian, 1);
            $hariCadangan = 10;
            $jumlahDisarankan = max(0, ceil(($rataRataHarian * $hariCadangan) - $pakan->stok_saat_ini));

            foreach ($aturanPakan as $aturan) {
                $terpenuhi = $this->cekKondisiNumerik(
                    $sisaHariStok,
                    $aturan->operator,
                    $aturan->nilai_ambang,
                    $aturan->nilai_ambang_atas
                );

                // Bersihkan rekomendasi aktif lama untuk kombinasi aturan + pakan ini agar tidak menumpuk
                Rekomendasi::where('kandang_id', $kandang->id)
                    ->where('aturan_rekomendasi_id', $aturan->id)
                    ->where('status', 'aktif')
                    ->where('alasan_data', 'like', "%{$pakan->nama_pakan}%")
                    ->delete();

                if ($terpenuhi) {
                    $alasan = "Pakan: {$pakan->nama_pakan} | Sisa stok: {$pakan->stok_saat_ini} {$pakan->satuan} | "
                        . "Rata-rata pakan/hari: {$rataRataHarian} {$pakan->satuan} | Sisa hari stok: {$sisaHariStok} hari | "
                        . "Disarankan beli: {$jumlahDisarankan} {$pakan->satuan} (cadangan {$hariCadangan} hari).";

                    Rekomendasi::create([
                        'kandang_id' => $kandang->id,
                        'aturan_rekomendasi_id' => $aturan->id,
                        'tanggal' => Carbon::today(),
                        'nama_aturan' => "{$aturan->nama_aturan} ({$pakan->nama_pakan})",
                        'isi_rekomendasi' => $aturan->saran_tindakan,
                        'alasan_data' => $alasan,
                        'prioritas' => $aturan->prioritas,
                        'status' => 'aktif',
                    ]);
                }
            }
        }
    }

    /**
     * Evaluasi Aturan Kesehatan (R4, R5, R6, R7, dan aturan kesehatan kustom)
     */
    public function evaluasiKesehatan(Kandang $kandang): void
    {
        $catatanTerbaru = CatatanHarian::where('kandang_id', $kandang->id)
            ->orderByDesc('tanggal')
            ->orderByDesc('id')
            ->take(3)
            ->get();

        if ($catatanTerbaru->isEmpty()) {
            return;
        }

        $hariIni = $catatanTerbaru->first();
        $aturanKesehatan = AturanRekomendasi::where('kategori', 'kesehatan')
            ->where('aktif', true)
            ->get();

        // Hitung persentase kematian harian
        $persenKematian = $hariIni->jumlah_ayam > 0
            ? round(($hariIni->kematian / $hariIni->jumlah_ayam) * 100, 2)
            : 0;

        // Cek tren kematian meningkat 3 hari berturut-turut
        $trenMeningkat3Hari = false;
        if ($catatanTerbaru->count() === 3) {
            $h0 = $catatanTerbaru[0]->kematian;
            $h1 = $catatanTerbaru[1]->kematian;
            $h2 = $catatanTerbaru[2]->kematian;
            if ($h0 > $h1 && $h1 > $h2 && $h0 > 0) {
                $trenMeningkat3Hari = true;
            }
        }

        foreach ($aturanKesehatan as $aturan) {
            $terpenuhi = false;
            $alasan = '';

            if ($aturan->parameter === 'persentase_kematian') {
                $terpenuhi = $this->cekKondisiNumerik($persenKematian, $aturan->operator, $aturan->nilai_ambang);
                $alasan = "Kandang: {$kandang->nama_kandang} | Kematian hari ini: {$hariIni->kematian} ekor dari {$hariIni->jumlah_ayam} ekor ({$persenKematian}%), melebihi ambang {$aturan->nilai_ambang}%.";
            } elseif ($aturan->parameter === 'tren_kematian_3_hari' && $trenMeningkat3Hari) {
                $terpenuhi = true;
                $alasan = "Kandang: {$kandang->nama_kandang} | Kematian meningkat 3 hari berturut-turut ({$catatanTerbaru[2]->kematian} → {$catatanTerbaru[1]->kematian} → {$catatanTerbaru[0]->kematian} ekor).";
            } elseif ($aturan->parameter === 'kondisi_kandang' && $aturan->kata_kunci_kondisi) {
                if (stripos($hariIni->kondisi, $aturan->kata_kunci_kondisi) !== false) {
                    $terpenuhi = true;
                    $alasan = "Kandang: {$kandang->nama_kandang} | Tercatat kondisi '{$hariIni->kondisi}' pada tanggal {$hariIni->tanggal}.";
                }
            }

            if ($terpenuhi) {
                Rekomendasi::updateOrCreate(
                    [
                        'kandang_id' => $kandang->id,
                        'aturan_rekomendasi_id' => $aturan->id,
                        'tanggal' => $hariIni->tanggal,
                        'status' => 'aktif',
                    ],
                    [
                        'nama_aturan' => $aturan->nama_aturan,
                        'isi_rekomendasi' => $aturan->saran_tindakan,
                        'alasan_data' => $alasan,
                        'prioritas' => $aturan->prioritas,
                    ]
                );
            }
        }
    }

                'kandang_id' => $riwayatBaru->kandang_id,
                'aturan_rekomendasi_id' => $aturanR7?->id,
                'tanggal' => $riwayatBaru->tanggal,
                'nama_aturan' => "Riwayat Penyakit Berulang: {$riwayatBaru->gejala_penyakit}",
                'isi_rekomendasi' => "{$saran} Tindakan terdahulu: {$riwayatSebelumnya->tindakan}",
                'alasan_data' => "Kandang: {$kandang->nama_kandang} | Gejala '{$riwayatBaru->gejala_penyakit}' pernah tercatat pada {$riwayatSebelumnya->tanggal}.",
                'prioritas' => $aturanR7 ? $aturanR7->prioritas : 'Waspada',
                'status' => 'aktif',
            ]);
        }
    }

    private function cekKondisiNumerik(float $nilai, string $operator, ?float $ambang, ?float $ambangAtas = null): bool
    {
        if ($ambang === null) {
            return false;
        }

        return match ($operator) {
            '<=' => $nilai <= $ambang,
            '<'  => $nilai < $ambang,
            '>=' => $nilai >= $ambang,
            '>'  => $nilai > $ambang,
            '==' => $nilai == $ambang,
            'between' => $ambangAtas !== null && ($nilai >= $ambang && $nilai <= $ambangAtas),
            default => false,
        };
    }
}

        function ubahQty(id, delta) {
            const input = document.getElementById('qty-' + id);
            const max = parseInt(input.getAttribute('max'));
            let val = parseInt(input.value || 0) + delta;
            if (val < 0) val = 0;
            if (val > max) val = max;
            input.value = val;
            hitungTotal();
        }

        function tambahSatu(id) {
            const input = document.getElementById('qty-' + id);
            if (parseInt(input.value || 0) === 0) {
                ubahQty(id, 1);
            }
        }

        function hitungTotal() {
            const inputs = document.querySelectorAll('.qty-input');
            const ringkasanEl = document.getElementById('daftar-ringkasan');
            let html = '';
            let total = 0;

            inputs.forEach(input => {
                const qty = parseInt(input.value || 0);
                if (qty > 0) {
                    const nama = input.dataset.nama;
                    const harga = parseFloat(input.dataset.harga);
                    const subtotal = qty * harga;
                    total += subtotal;
                    html += `<div class="d-flex justify-content-between mb-2">
                                <span>${nama} (${qty}x)</span>
                                <span class="fw-semibold">Rp ${subtotal.toLocaleString('id-ID')}</span>
                             </div>`;
                }
            });

            ringkasanEl.innerHTML = html || '<p class="text-muted small">Belum ada produk dipilih. Gunakan tombol (+) pada katalog di atas.</p>';
            document.getElementById('total-harga').innerText = 'Rp ' + total.toLocaleString('id-ID');
        }
